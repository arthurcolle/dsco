// Which reports need recompute after a curve update?
MATCH (s:Source {id: 'SOFR:2026-07-03'})
  -[:FEEDS]->(m:Model)
  -[:PRODUCES]->(o:Output)
WHERE o.status <> 'replayed'
RETURN
  o.id AS output,
  m.version AS model,
  shortestPath((s)-[*..4]->(o)) AS proof_path
ORDER BY o.priority DESC
